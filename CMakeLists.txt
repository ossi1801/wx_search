cmake_minimum_required(VERSION 3.16)
project(searchwx VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# wx-config lists Windows system libraries by name; expose the cross-compiler's
# own search directories to FindwxWidgets' library validation.
if(MINGW AND CMAKE_CROSSCOMPILING)
    list(APPEND CMAKE_LIBRARY_PATH ${CMAKE_CXX_IMPLICIT_LINK_DIRECTORIES})
endif()
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
find_package(Threads REQUIRED)
include(${wxWidgets_USE_FILE})
add_library(explorer_browser STATIC browser/explorer_frame.cpp browser/archives.cpp)
if(WIN32)
    target_sources(explorer_browser PRIVATE platform/windows/services.cpp)
else()
    target_sources(explorer_browser PRIVATE platform/portable/services.cpp)
endif()
target_include_directories(explorer_browser PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(explorer_browser PUBLIC ${wxWidgets_LIBRARIES} Threads::Threads)
add_executable(searchwx WIN32 main.cpp)
if(WIN32)
    enable_language(RC)
    target_sources(searchwx PRIVATE resources/windows.rc)
endif()
target_link_libraries(searchwx PRIVATE explorer_browser)
if(MSVC)
    target_compile_options(explorer_browser PRIVATE /W4)
    target_compile_options(searchwx PRIVATE /W4)
else()
    target_compile_options(explorer_browser PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(searchwx PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(archive_tests tests/archive_tests.cpp browser/archives.cpp)
    target_include_directories(archive_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(archive_tests PRIVATE ${wxWidgets_LIBRARIES})
    add_test(NAME archive_tests COMMAND archive_tests)
    add_executable(filesystem_tests tests/filesystem_tests.cpp)
    target_include_directories(filesystem_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME filesystem_tests COMMAND filesystem_tests)
    add_executable(search_model_tests tests/search_model_tests.cpp)
    target_include_directories(search_model_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME search_model_tests COMMAND search_model_tests)
    add_executable(taskbar_model_tests tests/taskbar_model_tests.cpp)
    target_include_directories(taskbar_model_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME taskbar_model_tests COMMAND taskbar_model_tests)
    add_executable(background_image_tests tests/background_image_tests.cpp)
    target_include_directories(background_image_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME background_image_tests COMMAND background_image_tests)
endif()
install(TARGETS searchwx RUNTIME DESTINATION bin)

option(EXPLORER_GUI_TESTS "Build the interactive GUI smoke test (requires a display)" OFF)
if(BUILD_TESTING AND EXPLORER_GUI_TESTS)
    add_executable(gui_smoke tests/gui_smoke.cpp)
    target_link_libraries(gui_smoke PRIVATE explorer_browser)
    if(UNIX AND NOT APPLE)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(GTK3 QUIET IMPORTED_TARGET gtk+-3.0)
        endif()
        if(GTK3_FOUND)
            target_link_libraries(gui_smoke PRIVATE PkgConfig::GTK3)
            target_compile_definitions(gui_smoke PRIVATE EXPLORER_GTK_SNAPSHOTS)
        endif()
    endif()
    add_test(NAME gui_smoke COMMAND gui_smoke)
endif()

option(EXPLORER_WINDOWS_SHELL "Build the experimental Windows taskbar companion" OFF)
if(EXPLORER_WINDOWS_SHELL)
    if(NOT WIN32)
        message(FATAL_ERROR "EXPLORER_WINDOWS_SHELL requires a Windows target")
    endif()
    target_sources(searchwx PRIVATE shell/windows/main.cpp shell/windows/background.cpp shell/windows/spotlight.cpp)
    target_compile_definitions(searchwx PRIVATE EXPLORER_WINDOWS_SHELL)
    target_link_libraries(searchwx PRIVATE shell32 ole32 uuid dwmapi gdi32 user32 comctl32 windowscodecs)
    if(MSVC)
        target_compile_options(searchwx PRIVATE /W4)
    else()
        target_compile_options(searchwx PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(BUILD_TESTING AND EXPLORER_GUI_TESTS)
        add_test(NAME windows_shell_smoke COMMAND searchwx --smoke-test)
        set_tests_properties(windows_shell_smoke PROPERTIES RUN_SERIAL TRUE TIMEOUT 30)
    endif()
endif()
